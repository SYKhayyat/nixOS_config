(require '14a-seforim-core)
(require '14b-seforim-candidates)
;; `embark' is NOT required at load time: the action commands below are plain

;; ---------------------------------------------------------------------------
;; Daf Navigation
;; ---------------------------------------------------------------------------
;; ASSUMPTION: in a text/org sefer, a daf header looks roughly like
;; "דף <hebrew-number> עמוד <א/ב>".  The regex below is a best-effort match; if
;; your files label dapim differently, adjust `re' here.
(defun seforim-goto-daf ()
  "Navigate to a Talmudic daf/amud in the current buffer or PDF."
  (interactive)
  (let* ((daf  (read-number "Daf: " 2))
         (alef (seforim--h-c 0))
         (bet  (seforim--h-c 1))
         (amud (completing-read "Amud: " (list alef bet) nil t)))
    (cond
     ((derived-mode-p 'pdf-view-mode)
      (if (fboundp 'my/pdf-jump-to-daf)
          (funcall 'my/pdf-jump-to-daf daf amud)
        (pdf-view-goto-page
         (+ (* (1- daf) 2) (if (string= amud alef) 0 1)))))
     (t
      (let* ((h-daf  (seforim--num-to-heb daf))
             (s-daf  (seforim--h-s 3 16))     ; דף
             (s-amud (seforim--h-s 16 13 5 3)) ; עמוד
             (re     (concat s-daf "\\s-+"
                             (regexp-quote h-daf)
                             "\\s-+" s-amud
                             "[^0-9]+" (regexp-quote amud))))
        (goto-char (point-min))
        (if (re-search-forward re nil t)
            (progn
              (when (derived-mode-p 'org-mode) (org-reveal))
              (recenter))
          (message "Daf not found")))))))

;; Study Log
;; ---------------------------------------------------------------------------

(defun seforim--load-log ()
  "Load the study log from `seforim-study-log-path' into `seforim--log-list'."
  (setq seforim--log-list nil)
  (when (file-exists-p seforim-study-log-path)
    (condition-case nil
        (with-temp-buffer
          (insert-file-contents seforim-study-log-path)
          (setq seforim--log-list (read (current-buffer))))
      (error nil))))

(defun seforim-show-log ()
  "Browse the study log and reopen a previously studied sefer.
Looks the file up by name under the library (no plocate dependency)."
  (interactive)
  (seforim--load-log)
  (if (null seforim--log-list)
      (message "Study log is empty.")
    (let* ((fmt (mapcar (lambda (x)
                          (format "%s  (%s)"
                                  (car x)
                                  (format-time-string "%Y-%m-%d %H:%M" (cdr x))))
                        seforim--log-list))
           (sel (completing-read "Recently studied: " fmt nil t)))
      (when sel
        (let* ((name (car (split-string sel "  (")))
               (file (car (directory-files-recursively
                           (seforim--base-dir)
                           (concat "\\`" (regexp-quote name) "\\'")))))
          (if file
              (seforim--open-file file nil)
            (message "File not found in library: %s" name)))))))

;; ---------------------------------------------------------------------------
;; In-Buffer Niqqud-Ignoring Isearch
;; ---------------------------------------------------------------------------
;; Lets you C-c s and type plain Hebrew to land on pointed text in the buffer.
(defun seforim--niqqud-isearch-search-fun (string &optional bound noerror count)
  "Isearch backend: match STRING regardless of niqqud, honouring direction."
  (let ((regex (seforim--build-niqqud-regex string)))
    (if isearch-forward
        (re-search-forward regex bound noerror count)
      (re-search-backward regex bound noerror count))))

(defun seforim-isearch-forward ()
  "Start isearch forward that ignores Hebrew niqqud."
  (interactive)
  (let ((isearch-search-fun-function
         (lambda () #'seforim--niqqud-isearch-search-fun)))
    (isearch-forward)))

(defun seforim-isearch-backward ()
  "Start isearch backward that ignores Hebrew niqqud."
  (interactive)
  (let ((isearch-search-fun-function
         (lambda () #'seforim--niqqud-isearch-search-fun)))
    (isearch-backward)))

(define-minor-mode seforim-mode
  "Minor mode for library files. Provides niqqud-ignoring isearch."
  :lighter " Seforim"
  :keymap (let ((map (make-sparse-keymap)))
            (define-key map seforim-inbuffer-niqqud-key #'seforim-isearch-forward)
            map))

(defun seforim--auto-enable-mode ()
  "Enable `seforim-mode' when visiting a file under the library."
  (when (and buffer-file-name
             (string-prefix-p (seforim--base-dir)
                              (expand-file-name buffer-file-name)))
    (seforim-mode 1)))
(add-hook 'find-file-hook #'seforim--auto-enable-mode)

;; ---------------------------------------------------------------------------
;; Library Status  (C-c S ?)
;; ---------------------------------------------------------------------------
(defun seforim-status ()
  "Display seforim library status: paths, subdir count, tools, options."
  (interactive)
  (seforim--load-log)
  (let* ((dir (seforim--base-dir))
         (exists (file-directory-p dir))
         (subdirs (when exists (seforim--list-subdirs)))
         (log-count (length seforim--log-list))
         (tools '(("plocate" . "filename search (system)")
                  ("fd"      . "filename search (library)")
                  ("rg"      . "text search (.org)")
                  ("rga"     . "deep text search (+PDF)")
                  ("recoll"  . "advanced indexed search")))
         (tool-status (mapconcat
                       (lambda (pair)
                         (format "  %-9s %-28s %s"
                                 (car pair) (cdr pair)
                                 (if (executable-find (car pair)) "[OK]" "[MISSING]")))
                       tools "\n")))
    (with-help-window "*Seforim Status*"
      (princ "Seforim Library Status\n")
      (princ "======================\n\n")
      (princ (format "Library dir:  %s\n" dir))
      (princ (format "Exists:       %s\n" (if exists "yes" "NO")))
      (princ (format "Subdirs:      %d\n" (length subdirs)))
      (when subdirs
        (dolist (d subdirs)
          (princ (format "  - %s\n"
                         (file-name-nondirectory (directory-file-name d))))))
      ;; Otzaria (mefarshim) status -- only if 14e is loaded.
      (when (boundp 'seforim-otzaria-directory)
        (let* ((odir seforim-otzaria-directory)
               (ok (and odir (file-directory-p odir))))
          (princ (format "\nOtzaria dir:  %s\n" (or odir "(unset)")))
          (princ (format "Otzaria OK:   %s\n" (if ok "yes" "NO")))
          (when (fboundp 'seforim--otzaria-links-dir)
            (let ((ld (seforim--otzaria-links-dir)))
              (princ (format "Links dir:    %s (%s)\n"
                             (or ld "(unknown)")
                             (if (and ld (file-directory-p ld))
                                 (format "%d link files"
                                         (length (directory-files ld nil "_links\\.json\\'")))
                               "missing")))))
          (princ "Change with:  M-x seforim-set-otzaria-directory   (or setq seforim-otzaria-directory)\n")))
      (princ (format "\nStudy log:    %d entries  (%s)\n" log-count seforim-study-log-path))
      (princ (format "\nTools:\n%s\n" tool-status))
      (princ (format "\nNiqqud ignore: %s\n" (if seforim-niqqud-ignore "ON" "off")))
      (princ (format "Max results:   %d\n" seforim-max-results)))))

;; ---------------------------------------------------------------------------
;; Embark Actions on files
;; ---------------------------------------------------------------------------
;; These AUGMENT the default Embark file actions (they do not replace them).
;; From any file candidate (e.g. a `seforim-search-rg' result) press C-. then:
;;   N  -> open in a new tab       O -> open with the system application
;;   Y  -> copy the file's basename
;; Capital keys are chosen to avoid clobbering Embark's built-in file bindings;
;; adjust freely if they collide in your Embark version.
(defun seforim-embark-open-tab (file)
  "Open FILE in a new tab."
  (interactive "f")
  (let ((seforim--new-tab-flag t))
    (seforim--open-file file nil nil)))

(defun seforim-embark-open-external (file)
  "Open FILE with the system default application."
  (interactive "f")
  (seforim--open-file file nil t))

(defun seforim-embark-copy-name (file)
  "Copy FILE's basename to the kill ring."
  (interactive "f")
  (kill-new (file-name-nondirectory file))
  (message "Copied: %s" (file-name-nondirectory file)))

(defun seforim-embark-consult-line (file)
  "Open FILE and run `consult-line' inside it."
  (interactive "f")
  (find-file file)
  (consult-line))

;; Keymap used for the custom `seforim-file' / `recoll-result' Embark
;; categories (registered in 03-completion.el).  Kept so those references
;; resolve; the same actions are also grafted onto `embark-file-map' below so
;; they work on ordinary file candidates (e.g. search results).
(defvar seforim-embark-file-map
  (let ((map (make-sparse-keymap)))
    (define-key map (kbd "t")   #'seforim-embark-open-tab)
    (define-key map (kbd "C-o") #'seforim-embark-open-external)
    (define-key map (kbd "c")   #'seforim-embark-copy-name)
    (define-key map (kbd "s")   #'seforim-embark-consult-line)
    map)
  "Embark keymap for seforim file candidates.")

(with-eval-after-load 'embark
  (when (boundp 'embark-file-map)
    (define-key embark-file-map (kbd "N") #'seforim-embark-open-tab)
    (define-key embark-file-map (kbd "O") #'seforim-embark-open-external)
    (define-key embark-file-map (kbd "Y") #'seforim-embark-copy-name)))
