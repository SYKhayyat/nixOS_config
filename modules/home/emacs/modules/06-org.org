;; FOOTNOTES IN ORG  (incl. footnote-on-footnote for LaTeX export)
;; ═══════════════════════════════════════════════════════════════════════════
;; Org supports only FLAT footnotes.  The old code here tried to encode nesting
;; in the label ([fn:1:2:3]), but Org reads ':' as inline-definition syntax, so
;; it never produced nested footnotes -- and its `:override' advice on
;; `org-latex-footnote-reference' risked mangling ordinary footnotes (reused
;; refs, multi-paragraph/block content).  It has been removed; normal Org
;; footnotes now export the standard, correct way.
;;
;; TO NEST a footnote inside a footnote in LaTeX/PDF output, drop a bigfoot
;; second-series command as a raw LaTeX snippet INSIDE a normal footnote's
;; definition:
;;
;;     Text.[fn:1]
;;     [fn:1] Outer note. @@latex:\footnoteB{Inner note (level 2).}@@
;;
;; The `article-hebrew' class below loads bigfoot and declares footnote series
;; B..J, so \footnoteB .. \footnoteJ nest correctly.  Helper to insert one:

(defun my/org-insert-nested-footnote (level)
  "Insert a raw-LaTeX level-LEVEL footnote snippet for Org export.
1 = \\footnote, 2 = \\footnoteB, ... 10 = \\footnoteJ.  Place it INSIDE an
ordinary Org footnote definition to get a footnote-on-a-footnote in the PDF."
  (interactive "nFootnote level (1-10): ")
  (let ((cmd (if (<= level 1) "footnote"
               (format "footnote%c" (+ ?A (1- level))))))
    (insert (format "@@latex:\\%s{}@@" cmd))
    (search-backward "}@@")))

;; RTL footnote fix
(with-eval-after-load 'org
  (advice-add 'org-footnote-new :after
              (lambda (&rest _)
                (when (eq bidi-paragraph-direction 'right-to-left)
                  (insert ?\x200F)))))

(defun my/org-fix-all-footnotes-rtl ()
  "Fix all footnotes for RTL."
  (interactive)
  (save-excursion
    (goto-char (point-min))
    (let ((count 0))
      (while (re-search-forward "^\\(\\[fn:[^]]+\\]\\)" nil t)
        (unless (looking-at (char-to-string ?\x200F))
          (insert ?\x200F)
          (cl-incf count)))
      (message "Fixed %d footnotes" count))))

(global-set-key (kbd "C-c h f") #'my/org-fix-all-footnotes-rtl)

;; Auto-tangle config
(defun my/tangle-config ()
  "Tangle config.org when saved."
  (when (string-equal (buffer-file-name)
                      (expand-file-name "config.org" user-emacs-directory))
    (let ((org-confirm-babel-evaluate nil))
      (org-babel-tangle))))

(add-hook 'org-mode-hook
          (lambda ()
            (add-hook 'after-save-hook #'my/tangle-config nil t)))
