;;; 10-context.el --- ConTeXt mode configuration -*- lexical-binding: t; -*-

;; ConTeXt mode (built into AUCTeX)
(with-eval-after-load 'tex
  (add-to-list 'auto-mode-alist '("\\.mkiv\\'" . ConTeXt-mode))
  (add-to-list 'auto-mode-alist '("\\.mkxl\\'" . ConTeXt-mode))
  (add-to-list 'auto-mode-alist '("\\.mkii\\'" . ConTeXt-mode)))

(add-hook 'ConTeXt-mode-hook #'my/setup-rtl-mode)
(add-hook 'ConTeXt-mode-hook #'visual-line-mode)
(add-hook 'ConTeXt-mode-hook #'electric-pair-local-mode)

(defun my/context-compile ()
  "Compile current ConTeXt file."
  (interactive)
  (compile (format "context %s" (shell-quote-argument buffer-file-name))))

(defun my/context-view ()
  "View compiled PDF."
  (interactive)
  (let ((pdf (concat (file-name-sans-extension buffer-file-name) ".pdf")))
    (if (file-exists-p pdf)
        (find-file-other-window pdf)
      (message "PDF not found"))))
