(defun my/gematria-calculate (string)
  "Calculate the gematria of STRING."
  (interactive "sHebrew word: ")
  (let ((values '((?א . 1) (?ב . 2) (?ג . 3) (?ד . 4) (?ה . 5) (?ו . 6)
                  (?ז . 7) (?ח . 8) (?ט . 9) (?י . 10) (?כ . 20) (?ך . 20)
                  (?ל . 30) (?מ . 40) (?ם . 40) (?נ . 50) (?ן . 50) (?ס . 60)
                  (?ע . 70) (?פ . 80) (?ף . 80) (?צ . 90) (?ץ . 90) (?ק . 100)
                  (?ר . 200) (?ש . 300) (?ת . 400)))
        (total 0))
    (dolist (char (append string nil))
      (setq total (+ total (or (cdr (assoc char values)) 0))))
    (if (called-interactively-p 'any)
        (message "Gematria of %s is: %d" string total)
      total)))

(defun my/insert-hebrew-date ()
  "Insert the current Hebrew date."
  (interactive)
  (if (executable-find "hdate")
      (let ((date (shell-command-to-string "hdate -H --notime")))
        (insert (string-trim date)))
    (message "hdate not installed")))

;; Keybindings for scholarship
(global-set-key (kbd "C-c h G") #'my/gematria-calculate)
(global-set-key (kbd "C-c h D") #'my/insert-hebrew-date)
