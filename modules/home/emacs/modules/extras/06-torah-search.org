#+TITLE: Torah web search
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Sefaria and Otzar HaChochma

=essentials/21-web-search= sets up =engine-mode= with the general engines
(Google Scholar, nixpkgs, DuckDuckGo) under the =C-c /= prefix. This adds the
two that are only useful if you learn.

#+begin_src emacs-lisp
;;; 06-torah-search.el --- Sefaria and Otzar search engines -*- lexical-binding: t; -*-

(with-eval-after-load 'engine-mode
  ;; Sefaria
  (defengine sefaria
    "https://www.sefaria.org/search?q=%s"
    :keybinding "s")

  ;; Otzar HaChochma
  (defengine otzar
    "https://www.otzar.org/wotzar/store/otzar.aspx?find=%s"
    :keybinding "o"))

(provide '06-torah-search)
;;; 06-torah-search.el ends here
#+end_src
