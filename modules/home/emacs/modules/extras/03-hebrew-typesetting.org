#+TITLE: Hebrew preambles for LaTeX, Typst and ConTeXt
#+PROPERTY: header-args:emacs-lisp :tangle yes

* The minimum each backend needs to set Hebrew

=essentials/07-latex=, =08-typst= and =09-context= are plain, general editors
for those three formats. This module adds the Hebrew half: a preamble inserter
bound to =C-c p h= in each mode.

These are the *small* preambles — language, direction, fonts. The footnote
apparatus is a separate and much larger question; it lives in
[[file:05-rich-footnotes.org][05-rich-footnotes]] and binds =C-c p f=.

One recurring trap, recorded here because it has bitten twice: in Typst, RTL
follows from ~lang: "he"~. ~page(flipped: true)~ is *landscape orientation*, not
right-to-left, and setting it produces a sideways page that looks superficially
like a bidi problem.

#+begin_src emacs-lisp
;;; 03-hebrew-typesetting.el --- Hebrew preambles for the typesetting modes -*- lexical-binding: t; -*-

;; ── LaTeX ──────────────────────────────────────────────────────────────────
(defvar my/latex-hebrew-preamble
  "% ── Hebrew Support ──
\\usepackage{fontspec}
\\usepackage[bidi=basic]{babel}
\\babelprovide[main, import]{hebrew}
\\babelprovide[import]{english}
\\babelfont{rm}{David CLM}
\\babelfont{sf}{Nachlieli CLM}
\\babelfont{tt}{Miriam Mono CLM}"
  "LaTeX Hebrew preamble.  Needs lualatex or xelatex, not pdflatex.")

(defun my/insert-latex-hebrew-preamble ()
  "Insert LaTeX Hebrew preamble."
  (interactive)
  (insert my/latex-hebrew-preamble))

;; ── Typst ──────────────────────────────────────────────────────────────────
(defvar my/typst-hebrew-preamble
  "#set text(lang: \"he\", font: \"David CLM\")
// RTL follows from lang: \"he\" -- do NOT use page(flipped: true), that is landscape.
"
  "Typst Hebrew preamble.")

(defun my/insert-typst-hebrew-preamble ()
  "Insert Hebrew preamble for Typst."
  (interactive)
  (insert my/typst-hebrew-preamble))

;; ── ConTeXt ────────────────────────────────────────────────────────────────
(defvar my/context-hebrew-preamble
  "% ── Hebrew Support ──
\\mainlanguage[he]
\\setupalign[r2l]
\\definefontfamily[hebrew][rm][David CLM]
\\definefontfamily[hebrew][ss][Nachlieli CLM]
\\definefontfamily[hebrew][tt][Miriam Mono CLM]
\\setupbodyfont[hebrew,rm,11pt]
"
  "ConTeXt Hebrew preamble.")

(defun my/insert-context-hebrew-preamble ()
  "Insert Hebrew preamble for ConTeXt."
  (interactive)
  (insert my/context-hebrew-preamble))

;; ── Footnote series shortcuts (C-c f N) ────────────────────────────────────
;; These live here, not in essentials, because the series they insert only
;; exist once the rich-footnote preamble is in the document.  In a plain LaTeX
;; file \footnoteB is undefined, so binding them in the general editor would be
;; offering a key that produces a document that will not compile.
;;
;; LaTeX and ConTeXt get ten (\fn plus B..J); Typst gets five, because Typst has
;; five TIERS rather than ten series -- see 05-rich-footnotes.

(defun my/tex--footnote-macro (n)
  "Name of the Nth footnote macro: 1 -> \"fn\", 2 -> \"fnB\" ... 10 -> \"fnJ\"."
  (if (<= n 1) "fn" (format "fn%c" (+ ?A (1- n)))))

(defun my/define-footnote-keys (map n-series formatter)
  "Bind C-c f 1..N-SERIES in MAP to insert a footnote of that series.
FORMATTER takes the macro name and returns the text to insert; point is left
inside the braces.  Built in a loop rather than as N hand-written defuns --
the old version had thirty near-identical ones across three modules."
  (dotimes (i n-series)
    (let* ((n (1+ i))
           (key (format "C-c f %d" (if (= n 10) 0 n)))
           (macro (my/tex--footnote-macro n)))
      (define-key map (kbd key)
                  (lambda ()
                    (interactive)
                    (insert (funcall formatter macro))
                    (backward-char 1))))))

(with-eval-after-load 'latex
  (my/define-footnote-keys LaTeX-mode-map 10
                           (lambda (m) (format "\\%s{}" m))))

(with-eval-after-load 'context
  (when (boundp 'ConTeXt-mode-map)
    (my/define-footnote-keys ConTeXt-mode-map 10
                             (lambda (m) (format "\\%s{}" m)))))

(with-eval-after-load 'typst-ts-mode
  (when (boundp 'typst-ts-mode-map)
    (my/define-footnote-keys typst-ts-mode-map 5
                             (lambda (m) (format "#%s[]" m)))))

;; ── Keybindings ────────────────────────────────────────────────────────────
;; C-c p h -> the Hebrew preamble above.
;; C-c p f -> the rich-footnote preamble, defined in 05-rich-footnotes, which
;;            loads after this file.  A keybinding resolves its symbol when the
;;            key is pressed, so the order does not matter.
(with-eval-after-load 'latex
  (define-key LaTeX-mode-map (kbd "C-c p h") #'my/insert-latex-hebrew-preamble)
  (define-key LaTeX-mode-map (kbd "C-c p f") #'my/insert-latex-rich-footnotes-preamble))

(with-eval-after-load 'typst-ts-mode
  (when (boundp 'typst-ts-mode-map)
    (define-key typst-ts-mode-map (kbd "C-c p h") #'my/insert-typst-hebrew-preamble)
    (define-key typst-ts-mode-map (kbd "C-c p f") #'my/insert-typst-rich-footnotes-preamble)))

(with-eval-after-load 'context
  (when (boundp 'ConTeXt-mode-map)
    (define-key ConTeXt-mode-map (kbd "C-c p h") #'my/insert-context-hebrew-preamble)
    (define-key ConTeXt-mode-map (kbd "C-c p f") #'my/insert-context-rich-footnotes-preamble)))

(provide '03-hebrew-typesetting)
;;; 03-hebrew-typesetting.el ends here
#+end_src
