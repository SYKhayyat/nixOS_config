#+TITLE: Seforim integration points
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Where the library hooks into the base config

Small pieces that belong to the seforim system but have to reach into modules
that live in =essentials/=. They are gathered here so those modules stay
general, and so there is one place to look when a seforim feature seems to be
missing from a menu it should be in.

Each hook uses the extension point the essentials module exports rather than
redefining anything, so nothing here breaks if the base module changes shape.

#+begin_src emacs-lisp
;;; 16-seforim-integration.el --- Seforim hooks into the base config -*- lexical-binding: t; -*-

;; ── File manager ───────────────────────────────────────────────────────────
;; `a' in dirvish (quick access) gets a jump to the library.
(with-eval-after-load 'dirvish
  (add-to-list 'dirvish-quick-access-entries
               (list "s" (or (bound-and-true-p seforim-directory)
                             "~/Documents/seforim")
                     "Seforim")
               t))

;; ── Buffer list ────────────────────────────────────────────────────────────
;; essentials/04-navigation builds its ibuffer groups from this variable at
;; ibuffer-mode time, so pushing here is enough -- no need to rebuild the list.
(with-eval-after-load 'ibuffer
  (add-to-list 'my/ibuffer-extra-groups '("Seforim" (filename . "seforim"))))

;; ── PDF ────────────────────────────────────────────────────────────────────
;; A Gemara PDF is paginated by daf, not by page: daf 2a is the third page of
;; the file.  `G' asks for a daf and goes there; `g' (plain page number) is
;; still bound in essentials/11-pdf.
(defun my/pdf-daf-to-page (daf-num amud)
  "Return the PDF page showing DAF-NUM AMUD.
AMUD is \"א\" or \"ב\".  Daf 2 amud א is page 3, and each daf is two pages."
  (+ (* (- daf-num 2) 2)
     (if (string= amud "א") 3 4)))

(defun my/pdf-jump-to-daf (daf-num amud)
  "Jump this PDF buffer to DAF-NUM AMUD."
  (interactive "nDaf: \nsAmud (א/ב): ")
  (pdf-view-goto-page (my/pdf-daf-to-page daf-num amud)))

(with-eval-after-load 'pdf-view
  ;; `seforim-goto-daf' (from the seforim modules) knows how to do this for org
  ;; and PDF alike; fall back to the PDF-only version when it is not loaded.
  (define-key pdf-view-mode-map (kbd "G")
              (if (fboundp 'seforim-goto-daf) #'seforim-goto-daf #'my/pdf-jump-to-daf)))

(provide '16-seforim-integration)
;;; 16-seforim-integration.el ends here
#+end_src
