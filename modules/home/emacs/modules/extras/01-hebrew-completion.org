#+TITLE: Hebrew-aware completion
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Niqqud-insensitive matching

=essentials/02-completion= sets up Orderless with =my/orderless-dispatcher=,
which handles the two prefixes worth having everywhere (~-term~ to exclude,
~=term~ for a literal match). This module pushes a Hebrew dispatcher in *front*
of it, so an all-Hebrew pattern is handled here and everything else falls
through to the generic one untouched.

The point: you type plain, unvowelled Hebrew and it matches text that carries
niqqud, and vav/yod are optional so /ktiv male/ and /ktiv chaser/ spellings of
the same word both hit.

#+begin_src emacs-lisp
;;; 01-hebrew-completion.el --- Niqqud-insensitive completion -*- lexical-binding: t; -*-

(defun seforim-orderless-hebrew-dispatcher (pattern _index _total)
  "Orderless dispatcher for all-Hebrew PATTERN.
Builds a regexp that tolerates niqqud between letters and treats vav/yod as
optional (ktiv male/chaser spelling variants).  Returns nil for anything that
is not entirely Hebrew letters, which leaves the pattern to the dispatchers
and styles behind this one.

NOTE: an earlier version did (replace-regexp-in-string \"ו\" \"ו?\") and then
`regexp-quote'd EVERY character including the inserted ?, producing a pattern
that literally required a \"?\".  This builds the regexp correctly and hands it
to Orderless via the `orderless-regexp' style."
  (when (string-match-p "\\`[א-ת]+\\'" pattern)
    (let* ((niqqud (concat "[" (char-to-string #x0591) "-"
                           (char-to-string #x05C7) "]*"))
           (rx (mapconcat
                (lambda (c)
                  (let ((q (regexp-quote (char-to-string c))))
                    (if (memq c '(?ו ?י))
                        (concat q "?" niqqud)   ; optional vav/yod + niqqud
                      (concat q niqqud))))      ; letter + optional niqqud
                (append pattern nil)
                "")))
      (cons 'orderless-regexp (concat niqqud rx)))))

(with-eval-after-load 'orderless
  (add-to-list 'orderless-style-dispatchers #'seforim-orderless-hebrew-dispatcher))

;; ── Embark actions on seforim results ──────────────────────────────────────
;; The seforim modules define `seforim-embark-file-map' and the two commands
;; below.  Keeping the minibuffer open after them is what makes "open several
;; of these" work.
(with-eval-after-load 'embark
  (setq embark-quit-after-action
        (append '((seforim-embark-open-tab . nil)
                  (seforim-embark-copy-citation . nil))
                (bound-and-true-p embark-quit-after-action)))
  (add-to-list 'embark-keymap-alist '(seforim-file . seforim-embark-file-map))
  (add-to-list 'embark-keymap-alist '(recoll-result . seforim-embark-file-map)))

(provide '01-hebrew-completion)
;;; 01-hebrew-completion.el ends here
#+end_src
