* The menus for the library and for text direction

=essentials/23-hydras= loads =hydra= and defines the general menus. These two
belong to this half of the config: =C-c S= drives the seforim system and
=C-c D= drives RTL/LTR.

The Spell line that used to sit in the direction hydra is gone. Hebrew spell
checking through Enchant was not reliable enough to be worth the key, and
=essentials/23-hydras= now has a general =s: set language= under =C-c W=
for the languages that do work.

#+begin_src emacs-lisp
;;; 17-hydras.el --- Seforim and direction hydras -*- lexical-binding: t; -*-

;; Forward declaration: defined in the seforim modules, which load before this.
(defvar seforim-directory)

(with-eval-after-load 'hydra

  (defhydra hydra-seforim (:color blue :hint nil)
    "
+------------------------------------------------------------------------+
| S E F O R I M                                                          |
+------------------------------------------------------------------------+
| FIND FILES                          SEARCH TEXT (niqqud-insensitive)   |
|  f: plocate (system-wide)            s: rg   (.org library)            |
|  F: fd (exact, library)              S: rga  (deep, incl. PDFs)        |
|  z: fd fuzzy (library)                                                 |
+------------------------------------------------------------------------+
| MEFARSHIM (Otzaria corpus)          READING & NAVIGATION               |
|  o: OPEN Otzaria twin of this book   w: reading room toggle            |
|  m: mefaresh here   M: side-pane     e: open EPUB file                 |
|  T: follow-pane toggle   R: reader   d: jump to daf (PDF/Org)          |
|  x: search Otzaria   X: browse       i: consult outline   t: TOC       |
+------------------------------------------------------------------------+
| EXTRAS   H: dashboard   r: jump to reference   g: recoll search        |
|          k: bookmark here   j: jump to bookmark                        |
+------------------------------------------------------------------------+
| LIBRARY:  l: study log   b: browse library   ?: status                 |
+------------------------------------------------------------------------+
"
    ;; Find files
    ("f" seforim-find-plocate)
    ("F" seforim-find-fd)
    ("z" seforim-find-fuzzy)

    ;; Search text (both bind the niqqud-insensitive compiler)
    ("s" seforim-search-rg)
    ("S" seforim-search-rga)

    ;; Mefarshim (Otzaria) -- requires 14-seforim-mefarshim and
    ;; `seforim-otzaria-directory' pointing at the Otzaria corpus
    ("o" seforim-open-in-otzaria)   ; hybrid entry: jump into the Otzaria twin
    ("m" seforim-mefarshim)
    ("M" seforim-mefarshim-show)
    ("T" seforim-mefarshim-follow-mode)
    ("x" seforim-otzaria-search)
    ("X" seforim-otzaria-browse)

    ;; Reading & navigation
    ("w" my/toggle-reading-room)
    ("R" seforim-otzaria-reader-mode)   ; prettified reader for raw Otzaria .txt
    ("e" (lambda () (interactive)
           (find-file (read-file-name "Open EPUB: " (seforim--base-dir) nil t nil
                                      (lambda (f) (or (file-directory-p f)
                                                      (string-suffix-p ".epub" f t)))))))
    ("d" seforim-goto-daf)
    ("i" consult-outline)
    ("t" seforim-toc)                   ; per-book table-of-contents sidebar
    ("b" (lambda () (interactive) (dired (seforim--base-dir))))

    ;; 15-seforim-dream
    ("H" seforim-dashboard)             ; clickable library home screen
    ("r" seforim-goto-ref)              ; searchRefs-style reference jump
    ("g" seforim-recoll-search)         ; indexed full-text search (if recoll)
    ("k" seforim-bookmark-set)
    ("j" seforim-bookmark-jump)

    ;; Library management
    ("l" seforim-show-log)
    ("?" seforim-status)

    ("q" nil))

  (global-set-key (kbd "C-c S") #'hydra-seforim/body)

  (defhydra hydra-direction (:color blue :hint nil)
    "
Direction: r: RTL  l: LTR  t: toggle  B: bidi reorder
Hebrew:    i: toggle Hebrew input  n: new Hebrew document
Notes:     f: fix all footnotes for RTL
"
    ("r" my/set-rtl)
    ("l" my/set-ltr)
    ("t" my/toggle-direction)
    ("B" my/toggle-bidi-reordering)
    ("i" my/toggle-hebrew-input)
    ("n" my/create-hebrew-document)
    ("f" my/org-fix-all-footnotes-rtl)
    ("q" nil))

  (global-set-key (kbd "C-c D") #'hydra-direction/body))

(provide '17-hydras)
;;; 17-hydras.el ends here
#+end_src
