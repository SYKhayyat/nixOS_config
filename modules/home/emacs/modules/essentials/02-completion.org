;; Orderless - flexible completion style
(use-package orderless
  :demand t
  :config
  (defun my/orderless-dispatcher (pattern _index _total)
    "Orderless dispatcher for the two prefixes worth having everywhere.
-term  -> exclude,  =term  -> literal match.
Returns nil for anything else, which leaves the pattern to the normal styles.

`orderless-style-dispatchers' is a list and each entry is tried in turn, so a
script- or project-specific dispatcher can be pushed onto the front of it from
another module without touching this one."
    (cond
     ((string-prefix-p "-" pattern)
      `(orderless-without-literal . ,(substring pattern 1)))
     ((string-prefix-p "=" pattern)
      `(orderless-literal . ,(substring pattern 1)))))

  (setq completion-styles '(orderless basic)
        completion-category-defaults nil
        orderless-style-dispatchers '(my/orderless-dispatcher)
        completion-category-overrides '((file (styles partial-completion)))))

;; Jinx - spell checking
;;
;; Jinx compiles a small C module against Enchant.  Nix builds that for us; on
;; a bare machine package.el installs the elisp and then the module fails to
;; load.  Gate on the capability so a box without Enchant quietly loses spell
;; check instead of taking vertico/consult/corfu down with the rest of this
;; module.  `M-x my/report-capabilities' lists anything skipped.
(use-package jinx
  :if (my/package-usable-p "jinx" "enchant-2")
  :hook ((text-mode . jinx-mode)
         (org-mode . jinx-mode))
  :bind ("M-$" . jinx-correct)
  :config
  (setq jinx-languages "en_US"))

(defun my/spell-set-language (langs)
  "Set spell checking in this buffer to LANGS and turn `jinx-mode' on.
LANGS is a space-separated list of Enchant language tags, e.g. \"en_US\" or
\"en_US de_DE\".  Prompts with whatever dictionaries Enchant actually reports,
so the offered list is the truth on this machine rather than a hardcoded set."
  (interactive
   ;; No candidate list: jinx exposes no public API for \"what dictionaries does
   ;; Enchant have\", and guessing at an internal one is how you get a config
   ;; that breaks on the next release.  Free text, defaulted to the current
   ;; value, is honest and always works.
   (list (read-string "Spell language(s): " (bound-and-true-p jinx-languages))))
  (if (not (fboundp 'jinx-mode))
      (message "Spell check unavailable here (jinx needs Enchant + a C compiler)")
    (setq-local jinx-languages langs)
    (jinx-mode 1)
    (message "Spell: %s" langs)))
