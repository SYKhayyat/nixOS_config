#+TITLE: Universal Scholar Search

* Engine Mode for Scholars

#+begin_src emacs-lisp
;;; 21-web-search.el --- Search the web from Emacs -*- lexical-binding: t; -*-

(use-package engine-mode
  :demand t
  :config
  (engine-mode t)
  
  ;; Google Scholar
  (defengine scholar
    "https://scholar.google.com/scholar?q=%s"
    :keybinding "S")

  ;; Nix Packages (For your system)
  (defengine nixpkgs
    "https://search.nixos.org/packages?query=%s"
    :keybinding "p")

  ;; DuckDuckGo
  (defengine ddg
    "https://duckduckgo.com/?q=%s"
    :keybinding "d"))

;; Keybinding prefix: C-c / followed by the key above (e.g., C-c / s for Sefaria)
(setq engine-mode-keymap-prefix (kbd "C-c /"))

(provide '21-web-search)
;;; 21-web-search.el ends here
#+end_src
