;;; 11-pdf.el --- PDF and EPUB with NixOS store support -*- lexical-binding: t; -*-

;; pdf-tools needs a compiled helper program, `epdfinfo'.  How we obtain it
;; depends on the environment (see `my/portable-packages' in 00-core):
;;   * NIX MODE  -> Nix already built epdfinfo and ships it in the store next to
;;                  the pdf-tools elisp.  We must NOT let pdf-tools try to
;;                  recompile it (that fails), and we point it at the store copy.
;;   * PORTABLE  -> pdf-tools is installed from MELPA; run `M-x pdf-tools-install'
;;                  once to build epdfinfo (needs poppler + a C toolchain).
(defvar my/pdf-nix-mode (not (bound-and-true-p my/portable-packages))
  "Non-nil when pdf-tools comes from Nix (do not self-compile epdfinfo).")

(when my/pdf-nix-mode
  ;; --- "Vaccines": neutralize the self-compiling installers Nix doesn't need.
  (defun pdf-loader-install (&optional _arg) (interactive))
  (defun pdf-tools-install (&optional _arg) (interactive))
  (defun pdf-occur-global-minor-mode (&optional _arg) (interactive))
  ;; --- Resolve the real Nix store directory and point epdfinfo at it.
  (when (require 'pdf-tools nil t)
    (let* ((wrapped-path (locate-library "pdf-tools"))
           (real-path (when wrapped-path (file-truename wrapped-path)))
           (real-dir (when real-path (file-name-directory real-path))))
      (when (and real-dir (file-directory-p real-dir))
        (add-to-list 'load-path real-dir)
        (setq pdf-info-epdfinfo-program (expand-file-name "epdfinfo" real-dir))
        (setenv "PDF_TOOLS_EPDFINFO" pdf-info-epdfinfo-program)
        (require 'pdf-util nil t)
        (require 'pdf-info nil t)
        (require 'pdf-view nil t)
        (require 'pdf-cache nil t)))))

;; Is pdf-tools actually usable here?  In portable mode `epdfinfo' must be
;; built against poppler, so we take the presence of poppler's own `pdfinfo'
;; as the proxy.  Without it we must NOT claim .pdf in `auto-mode-alist', or
;; every PDF opens as a raw-bytes buffer that cannot be read -- worse than
;; letting Emacs use its built-in doc-view fallback.
(defvar my/pdf-tools-available (my/package-usable-p "pdf-tools" "pdfinfo")
  "Non-nil when pdf-tools can render PDFs on this machine.")

;; Open PDFs in pdf-view-mode (prevents raw-bytes "gibberish" buffers).
(when my/pdf-tools-available
  (setq auto-mode-alist (append '(("\\.pdf\\'" . pdf-view-mode)) auto-mode-alist)))

(use-package pdf-tools
  ;; No :ensure here on purpose: it inherits `use-package-always-ensure', so it
  ;; installs from MELPA in portable mode and uses Nix's copy otherwise.
  :if my/pdf-tools-available
  :demand t
  :config
  (condition-case err
      (when (fboundp 'pdf-info-initialize)
        (pdf-info-initialize)
        (pdf-cache-install)
        (pdf-history-install))
    (error
     (message "pdf-tools not ready: %s (run M-x pdf-tools-install if not on Nix)"
              (error-message-string err))))
  (setq-default pdf-view-display-size 'fit-page)
  (setq pdf-view-use-scaling t)
  (setq pdf-view-midnight-colors '("#dcdccc" . "#1a1a1a"))
  (add-hook 'pdf-view-mode-hook #'pdf-view-midnight-minor-mode))

;; Persist PDF position across sessions
(use-package pdf-view-restore
  :after pdf-tools
  :hook (pdf-view-mode . pdf-view-restore-mode))

;; EPUB reader
(use-package nov
  :mode ("\\.epub\\'" . nov-mode)
  :config
  (setq nov-save-place-mode t)
  (add-hook 'nov-mode-hook 'visual-line-mode))

(with-eval-after-load 'pdf-view
  (define-key pdf-view-mode-map (kbd "g") (lambda (p) (interactive "nPage: ") (pdf-view-goto-page p)))
  (define-key pdf-view-mode-map (kbd "o") 'pdf-outline)
  (define-key pdf-view-mode-map (kbd "s s") 'pdf-occur))

(provide '11-pdf)
#+end_src
