;;; 05-org.el --- Org mode settings -*- lexical-binding: t; -*-

(use-package org
  :hook ((org-mode . visual-line-mode))
  :config
  ;; Load export backends (Fixes Problem 1)
  (require 'ox-org)
  (require 'ox-latex)
  (require 'ox-md)

  ;; Basic settings
  (setq org-startup-indented t
        org-hide-leading-stars t
        org-hide-emphasis-markers t
        org-ellipsis " ▾"
        org-pretty-entities t
        org-src-fontify-natively t
        org-src-tab-acts-natively t
        org-confirm-babel-evaluate nil
        org-edit-src-content-indentation 0
        org-return-follows-link t
	org-footnote-auto-adjust t 
        org-log-done 'time)

  ;; Deep nesting support (20 levels)
  (setq org-n-level-faces 20
        org-cycle-level-faces nil
        org-list-allow-alphabetical t
        org-list-indent-offset 2
        org-list-demote-modify-bullet '(("+" . "-") ("-" . "+") ("*" . "-")))

  ;; TODO and tags
  (setq org-todo-keywords
        '((sequence "TODO(t)" "IN-PROGRESS(p)" "WAITING(w)" "|" "DONE(d)" "CANCELLED(c)"))
        org-directory "~/Documents/org/"
        org-agenda-files '("~/Documents/org/")
        ;; Extras modules append their own tags with `add-to-list'.
        org-tag-alist
        '(("research" . ?r) ("urgent" . ?u) ("writing" . ?w) ("reading" . ?R))))

;; Babel languages
(with-eval-after-load 'org
  (org-babel-do-load-languages
   'org-babel-load-languages
   '((emacs-lisp . t)
     (shell . t)
     (python . t)
     (latex . t))))

(use-package org-modern
  :hook (org-mode . org-modern-mode)
  :config
  (setq org-modern-star '("◉" "○" "◈" "◇" "✦" "✧" "✶" "✷" "❋" "❊"
                          "◉" "○" "◈" "◇" "✦" "✧" "✶" "✷" "❋" "❊")
        org-modern-hide-stars 'leading
        org-modern-table t))

;; Capture templates
(with-eval-after-load 'org
  (setq org-capture-templates
        '(("t" "Todo" entry (file+headline "~/Documents/org/inbox.org" "Tasks")
           "* TODO %?\n  %i\n  %a")
          ("n" "Note" entry (file+headline "~/Documents/org/inbox.org" "Notes")
           "* %?\n  %i\n  %a")
          ("j" "Journal" entry (file+datetree "~/Documents/org/journal.org")
           "* %?\n  Entered on %U\n  %i")
          ("r" "Research" entry (file+headline "~/Documents/org/research.org" "Inbox")
           "* %?\n  :PROPERTIES:\n  :SOURCE: \n  :END:\n  %i\n  %a"))))

(global-set-key (kbd "C-c a") #'org-agenda)
(global-set-key (kbd "C-c c") #'org-capture)

(use-package org-download
  :after org
  :hook (org-mode . org-download-enable)
  :config
  (setq org-download-method 'directory
        org-download-image-dir "./images"
        org-download-heading-lvl nil))

;; ═══════════════════════════════════════════════════════════════════════════
;; FOOTNOTES IN ORG
;; ═══════════════════════════════════════════════════════════════════════════
;; Org supports only FLAT footnotes.  The old code here tried to encode nesting
;; in the label ([fn:1:2:3]), but Org reads ':' as inline-definition syntax, so
;; it never produced nested footnotes -- and its `:override' advice on
;; `org-latex-footnote-reference' risked mangling ordinary footnotes (reused
;; refs, multi-paragraph/block content).  It has been removed; normal Org
;; footnotes now export the standard, correct way.

;; ═══════════════════════════════════════════════════════════════════════════
;; LATEX EXPORT
;; ═══════════════════════════════════════════════════════════════════════════
;; `org-latex-headline-levels' is left at Org's default.  It was 20 here, which
;; only meant levels 6-20 all became \subparagraph -- the deeper headings were
;; aliases, not levels.

(with-eval-after-load 'ox-latex
  (setq org-latex-compiler "lualatex")

  (setq org-latex-pdf-process
        '("latexmk -pdflatex='lualatex -shell-escape -interaction nonstopmode' -pdf -bibtex -f %f"))

  (setq org-latex-packages-alist
        '(("" "fontspec" t))))
