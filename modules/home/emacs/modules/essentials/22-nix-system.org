* NixOS and Home Manager Tools

#+begin_src emacs-lisp
;;; 22-nix-system.el --- Manage NixOS from Emacs -*- lexical-binding: t; -*-

(defun my/nixos-rebuild ()
  "Rebuild the NixOS system."
  (interactive)
  (compile "sudo nixos-rebuild switch"))

(defun my/home-manager-switch ()
  "Switch Home Manager configuration."
  (interactive)
  (compile "home-manager switch"))

(defun my/nix-garbage-collect ()
  "Clean up the Nix store."
  (interactive)
  (compile "nix-collect-garbage -d"))
