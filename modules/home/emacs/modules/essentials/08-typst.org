;;; 08-typst.el --- Typst mode configuration -*- lexical-binding: t; -*-

(use-package treesit-auto
  :demand t
  :config
  (setq treesit-auto-install 'prompt)
  (treesit-auto-add-to-auto-mode-alist 'all)
  (global-treesit-auto-mode 1))

(use-package typst-ts-mode
  :mode "\\.typ\\'"
  :hook ((typst-ts-mode . electric-pair-local-mode))
  :config
  (with-eval-after-load 'eglot
    (add-to-list 'eglot-server-programs '(typst-ts-mode . ("tinymist")))))

(defun my/typst-compile ()
  "Compile current Typst file."
  (interactive)
  (compile (format "typst compile %s" (shell-quote-argument buffer-file-name))))

(defun my/typst-watch ()
  "Watch and compile Typst file."
  (interactive)
  (async-shell-command (format "typst watch %s" (shell-quote-argument buffer-file-name))))

(defun my/typst-view ()
  "View compiled PDF."
  (interactive)
  (let ((pdf (concat (file-name-sans-extension buffer-file-name) ".pdf")))
    (if (file-exists-p pdf)
        (find-file-other-window pdf)
      (message "PDF not found"))))

;; ═══════════════════════════════════════════════════════════════════════════
;; LIVE SIDE-BY-SIDE PREVIEW
;; ═══════════════════════════════════════════════════════════════════════════
;; Source on the left, the compiled PDF in a right-hand window that updates
;; itself.  `typst watch' recompiles on every save (it is incremental, so this
;; is fast even on a long document), and the PDF buffer auto-reverts when the
;; file changes on disk.
;;
;; This is deliberately NOT the `typst-preview' package: that renders into a
;; *browser* over a websocket, which is a different window manager's problem
;; and does not survive a headless/SSH session.  A pdf-view buffer is a real
;; Emacs window -- it splits, it themes, and it works the same everywhere.
;; If you do want the browser renderer, `M-x typst-preview' is still there
;; (the package is installed); the two are independent.

(defcustom my/typst-preview-width 0.5
  "Fraction of the frame width used by the Typst preview side window."
  :type 'number :group 'typst)

(defvar-local my/typst-preview--proc nil
  "The `typst watch' process for this buffer, if any.")

(defun my/typst-preview--pdf-path ()
  "PDF that `typst watch' will produce for the current buffer."
  (concat (file-name-sans-extension buffer-file-name) ".pdf"))

(defun my/typst-preview--show-pdf (pdf)
  "Display PDF in a right-hand side window with auto-revert enabled."
  (let ((buf (find-file-noselect pdf t)))
    (with-current-buffer buf
      ;; The PDF is rewritten by `typst watch' constantly; revert silently and
      ;; without the "file has changed on disk" prompt.
      (setq-local revert-without-query (list ".*"))
      (auto-revert-mode 1))
    (display-buffer
     buf
     `(display-buffer-in-side-window
       (side          . right)
       (slot          . 0)
       (window-width  . ,my/typst-preview-width)
       (preserve-size . (t . nil))))
    buf))

(define-minor-mode my/typst-preview-mode
  "Compile this Typst buffer on every change and show the PDF beside it.
Runs `typst watch' for the file and displays the resulting PDF in an
auto-reverting side window."
  :lighter " TY>"
  (if my/typst-preview-mode
      (cond
       ((not buffer-file-name)
        (setq my/typst-preview-mode nil)
        (user-error "Save the file first -- typst watch needs a path"))
       ((not (executable-find "typst"))
        (setq my/typst-preview-mode nil)
        (user-error "`typst' not found -- install it: %s"
                    (pcase system-type
                      ('windows-nt "scoop install typst")
                      ('darwin     "brew install typst")
                      (_           "NixOS: already in home.packages | else: cargo install typst-cli"))))
       (t
        (let ((pdf (my/typst-preview--pdf-path)))
          (setq my/typst-preview--proc
                (start-process "typst-watch"
                               (format " *typst watch: %s*"
                                       (file-name-nondirectory buffer-file-name))
                               "typst" "watch" buffer-file-name pdf))
          ;; The first compile takes a moment; show the PDF once it exists
          ;; rather than opening an empty/missing file.
          (run-with-timer
           0.8 nil
           (lambda (pdf)
             (if (file-exists-p pdf)
                 (my/typst-preview--show-pdf pdf)
               (message "Typst: waiting for first compile (see the *typst watch* buffer for errors)")))
           pdf)
          (message "Typst preview: watching %s"
                   (file-name-nondirectory buffer-file-name)))))
    ;; Disabling: stop the watcher and put the window away.
    (when (process-live-p my/typst-preview--proc)
      (delete-process my/typst-preview--proc))
    (setq my/typst-preview--proc nil)
    (when buffer-file-name
      (let* ((pdf (my/typst-preview--pdf-path))
             (buf (find-buffer-visiting pdf))
             (win (and buf (get-buffer-window buf))))
        (when (window-live-p win) (delete-window win))))))

(with-eval-after-load 'typst-ts-mode
  (when (boundp 'typst-ts-mode-map)
    (define-key typst-ts-mode-map (kbd "C-c C-c") #'my/typst-compile)
    (define-key typst-ts-mode-map (kbd "C-c C-w") #'my/typst-watch)
    (define-key typst-ts-mode-map (kbd "C-c C-v") #'my/typst-view)
    ;; Same key shape as the Markdown preview (C-c m p) so the two feel alike.
    (define-key typst-ts-mode-map (kbd "C-c m p") #'my/typst-preview-mode)))
