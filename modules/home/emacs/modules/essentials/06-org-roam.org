;;; 06-org-roam.el --- Org-roam for notes -*- lexical-binding: t; -*-

(use-package org-roam
  :demand t
  :custom
  (org-roam-directory "~/Documents/roam/")
  (org-roam-completion-everywhere t)
  (org-roam-dailies-directory "daily/")
  :bind (("C-c n f" . org-roam-node-find)
         ("C-c n i" . org-roam-node-insert)
         ("C-c n l" . org-roam-buffer-toggle)
         ("C-c n c" . org-roam-capture)
         ("C-c n g" . org-roam-graph)
         ("C-c n t" . org-roam-tag-add)
         ("C-c n d" . org-roam-dailies-goto-today)
         ("C-c n D" . org-roam-dailies-goto-date))
  :config
  (org-roam-db-autosync-mode 1)

  (setq org-roam-capture-templates
        '(("d" "default" plain "%?"
           :target (file+head "%<%Y%m%d%H%M%S>-${slug}.org"
                              "#+title: ${title}\n#+date: %U\n#+filetags: \n\n")
           :unnarrowed t)
          ("r" "research" plain "%?"
           :target (file+head "%<%Y%m%d%H%M%S>-${slug}.org"
                              "#+title: ${title}\n#+date: %U\n#+filetags: research\n\n* Source\n\n* Notes\n\n")
           :unnarrowed t)))

  (setq org-roam-dailies-capture-templates
        '(("d" "default" entry "* %<%H:%M> %?"
           :target (file+head "%<%Y-%m-%d>.org"
                              "#+title: %<%Y-%m-%d>\n#+filetags: daily\n\n")))))

(use-package org-roam-ui
  :after org-roam
  :config
  (setq org-roam-ui-sync-theme t
        org-roam-ui-follow t
        org-roam-ui-update-on-save t)
  :bind ("C-c n u" . org-roam-ui-mode))

(defun my/org-roam-find-tagged (tag)
  "Find org-roam nodes carrying TAG.
Generalises the one-command-per-tag pattern this used to have: any tag you
start using is immediately reachable without editing this file."
  (interactive
   (list (completing-read "Tag: " (org-roam-tag-completions) nil t)))
  (org-roam-node-find nil nil
                      (lambda (node) (member tag (org-roam-node-tags node)))))

(global-set-key (kbd "C-c n R") #'my/org-roam-find-tagged)

(provide '06-org-roam)
