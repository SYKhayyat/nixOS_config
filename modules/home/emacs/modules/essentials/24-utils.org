;; Config management
;; This used to open "config.org" in `user-emacs-directory'.  That file has
;; never existed in this layout -- the config is init.el plus modules/*.org --
;; so C-c e i opened an empty buffer named after a file that was never written.
(defun my/open-config ()
  "Open init.el, the entry point of this configuration."
  (interactive)
  (find-file (expand-file-name "init.el" user-emacs-directory)))

(defun my/reload-config ()
  "Reload the Emacs configuration."
  (interactive)
  (load-file (expand-file-name "init.el" user-emacs-directory))
  (message "Config reloaded"))

(defun my/open-modules ()
  "Open the literate modules directory in dired.
Opens `my/modules-root', which holds the essentials/ and extras/ groups."
  (interactive)
  (cond ((boundp 'my/modules-root) (dired my/modules-root))
        ((boundp 'my/modules-dir)  (dired my/modules-dir))
        (t (user-error "Module directory is not defined"))))

(global-set-key (kbd "C-c e i") #'my/open-config)
(global-set-key (kbd "C-c e r") #'my/reload-config)
(global-set-key (kbd "C-c e m") #'my/open-modules)

;; Auto-tangle a module's .org back to .el on save (modules only).
(defun my/auto-tangle-module ()
  "Re-tangle the current buffer if it is a literate module .org file.
Only fires for files under one of `my/modules-dirs' so unrelated .org saves
(notes, seforim, roam) are never tangled.  Checking every group directory
matters: testing only the first one would silently stop re-tangling extras/."
  (when (and buffer-file-name
             (string-suffix-p ".org" buffer-file-name)
             (let ((file (expand-file-name buffer-file-name))
                   (dirs (if (boundp 'my/modules-dirs)
                             my/modules-dirs
                           (and (boundp 'my/modules-dir) (list my/modules-dir)))))
               (seq-some (lambda (d) (string-prefix-p (expand-file-name d) file))
                         dirs)))
    (let ((org-confirm-babel-evaluate nil))
      (org-babel-tangle)
      (message "Tangled %s" (file-name-nondirectory buffer-file-name)))))

(add-hook 'org-mode-hook
          (lambda ()
            (add-hook 'after-save-hook #'my/auto-tangle-module nil t)))

(use-package restart-emacs
  :bind ("C-c e R" . restart-emacs))

;; Window splitting
(defun my/split-and-follow-h ()
  "Split window horizontally and move to new window."
  (interactive)
  (split-window-below)
  (balance-windows)
  (other-window 1))

(defun my/split-and-follow-v ()
  "Split window vertically and move to new window."
  (interactive)
  (split-window-right)
  (balance-windows)
  (other-window 1))

(global-set-key (kbd "C-x 2") #'my/split-and-follow-h)
(global-set-key (kbd "C-x 3") #'my/split-and-follow-v)

;; Server
(use-package server
  :ensure nil
  :config
  (unless (server-running-p)
    (server-start)))

;; Startup message
(add-hook 'emacs-startup-hook
          (lambda ()
            (message "Emacs ready in %.2f seconds"
                     (float-time (time-subtract after-init-time before-init-time)))))
