* Bibliography and Citations

;;; 19-academic.el --- Bibliography management -*- lexical-binding: t; -*-

(use-package citar
  :no-require t
  :bind (("C-c b b" . citar-open)
         :map org-mode-map
         ("C-c b i" . citar-insert-citation)
         ("C-c b r" . citar-insert-reference))
  :custom
  (citar-bibliography '("~/Documents/bibliography.bib"))
  (citar-notes-paths '("~/Documents/roam/"))
  :config
  (setq citar-symbols
        `((citation . (,(nerd-icons-faicon "nf-fa-quote_left") . " "))
          (ref . (,(nerd-icons-faicon "nf-fa-book") . " "))
          (note . (,(nerd-icons-faicon "nf-fa-sticky_note") . " "))
          (link . (,(nerd-icons-faicon "nf-fa-link") . " "))))
  (setq citar-symbol-separator "  "))

(use-package citar-org-roam
  :after (citar org-roam)
  :config
  (citar-org-roam-mode 1)
  (setq citar-org-roam-note-title-template "${author} - ${title}")
  ;; Match your research template
  (setq citar-org-roam-capture-template-spec
        '(:target (file+head "%<%Y%m%d%H%M%S>-${citekey}.org"
                            "#+title: ${author} - ${title}\n#+filetags: research\n\n* Source\n\n* Notes\n\n"))))

(provide '19-academic)
;;; 19-academic.el ends here
#+end_src
