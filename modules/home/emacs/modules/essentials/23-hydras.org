* Discoverable menus

Hydra turns a prefix key into a menu you can read, so nothing here has to be
memorised. This module loads =hydra= and defines the menus that belong to the
base config; =extras/17-hydras.el= adds its own on top.

The previous version of this file carried two comments — ~;; ... rest of hydras
unchanged ...~ and ~;; ... etc. (other hydras unchanged)~ — which read as though
more menus were defined further down. They were not: the file has only ever
defined three hydras. The placeholders are gone.

#+begin_src emacs-lisp
;;; 23-hydras.el --- Hydra menus -*- lexical-binding: t; -*-

(use-package hydra
  :demand t
  :config

  (defhydra hydra-roam (:color blue :hint nil)
    "
Roam: f: find node  i: insert link  c: capture
      d: daily  g: graph  u: UI toggle  t: by tag
"
    ("f" org-roam-node-find)
    ("i" org-roam-node-insert)
    ("c" org-roam-capture)
    ("d" org-roam-dailies-goto-today)
    ("g" org-roam-graph)
    ("u" org-roam-ui-mode)
    ("t" my/org-roam-find-tagged)
    ("q" nil))

  (global-set-key (kbd "C-c C-n") #'hydra-roam/body)

  (defhydra hydra-writing (:color blue :hint nil)
    "
Writing:  w: reading room (olivetti)   W: set width   f: focus mode
Spell:    s: set language              $: correct word at point
"
    ("w" my/toggle-reading-room)
    ("W" olivetti-set-width)
    ("f" focus-mode)
    ("s" my/spell-set-language)
    ("$" jinx-correct)
    ("q" nil))

  (global-set-key (kbd "C-c W") #'hydra-writing/body))

(provide '23-hydras)
;;; 23-hydras.el ends here
#+end_src
