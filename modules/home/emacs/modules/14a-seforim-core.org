#+TITLE: 14a Seforim Core – Config, Hebrew, Paths, Niqqud
#+PROPERTY: header-args:emacs-lisp :tangle 14a-seforim-core.el

Core of the seforim system: user options, Hebrew-numeral utilities, niqqud
handling (including the niqqud-insensitive Consult regexp compiler that makes
full-text search work), and path helpers that auto-detect the library
directory regardless of "Seforim"/"seforim" casing (NixOS is case-sensitive).

(defun seforim--default-directory ()
  "Return the first existing candidate seforim directory.
Tries capital and lowercase spellings so the config works on
case-sensitive filesystems without manual tweaking."
  (seq-find #'file-directory-p
            (list (expand-file-name "~/Documents/seforim/")
                  (expand-file-name "~/Documents/Seforim/")
                  (expand-file-name "~/seforim/"))
            (expand-file-name "~/Documents/seforim/")))

;; ---------------------------------------------------------------------------
;; Hebrew Numeral Utilities
;; ---------------------------------------------------------------------------

(defun seforim--h-c (id)
  "Hebrew char from Aleph offset (1488)."
  (char-to-string (+ 1488 id)))

(defun seforim--h-s (&rest ids)
  "Hebrew string from IDS."
  (apply #'concat (mapcar #'seforim--h-c ids)))

(defun seforim--num-to-heb (n)
  "Convert integer N to Hebrew numeral string."
  (let* ((thousands (/ n 1000))
         (rem       (% n 1000))
         (h-list    '(nil (23) (24) (25) (26) (26 23)
                      (26 24) (26 25) (26 26) (26 26 23)))
         (t-list    '(nil 9 10 11 12 13 14 18 21 22))
         (o-list    '(nil 0 1 2 3 4 5 6 7 8))
         (h-idx     (/ rem 100))
         (t-rem     (% rem 100))
         (t-idx     (/ t-rem 10))
         (o-idx     (% t-rem 10))
         (suffix
          (cond
           ((= t-rem 15) (seforim--h-s 8 5))
           ((= t-rem 16) (seforim--h-s 8 6))
           (t (concat
               (if (> t-idx 0) (seforim--h-c (nth t-idx t-list)) "")
               (if (> o-idx 0) (seforim--h-c (nth o-idx o-list)) ""))))))
    (concat
     (when (> thousands 0)
       (concat (seforim--h-c (nth thousands o-list)) "'"))
     (let ((codes (nth h-idx h-list)))
       (if (consp codes) (apply #'seforim--h-s codes) ""))
     suffix)))

;; ---------------------------------------------------------------------------
;; Niqqud Handling
;; ---------------------------------------------------------------------------

(defconst seforim--niqqud-lo #x0591 "First codepoint of the niqqud/te'amim block.")
(defconst seforim--niqqud-hi #x05C7 "Last codepoint of the niqqud/te'amim block.")
(defconst seforim--letter-lo #x05D0 "Hebrew letter Aleph.")
(defconst seforim--letter-hi #x05EA "Hebrew letter Tav (incl. final forms).")

(defconst seforim--niqqud-re
  (concat "[" (char-to-string seforim--niqqud-lo) "-"
          (char-to-string seforim--niqqud-hi) "]")
  "Regex matching a single Hebrew niqqud/cantillation character.")

(defconst seforim--niqqud-opt
  (concat seforim--niqqud-re "*")
  "Regex matching zero or more niqqud characters.
Uses literal Unicode chars, so it is valid in both ERE and PCRE.")

(defun seforim--strip-niqqud (str)
  "Remove Hebrew niqqud (U+0591-U+05C7) from STR."
  (replace-regexp-in-string seforim--niqqud-re "" str))

(defun seforim--letter-p (ch)
  "Non-nil if CH is a Hebrew letter (incl. final forms)."
  (<= seforim--letter-lo ch seforim--letter-hi))

(defun seforim--niqqud-char-p (ch)
  "Non-nil if CH is a Hebrew niqqud/cantillation mark."
  (<= seforim--niqqud-lo ch seforim--niqqud-hi))

(defun seforim--inject-niqqud (regexp)
  "Make REGEXP niqqud-insensitive.
Drop any niqqud already present in REGEXP, and after every Hebrew
letter insert an optional-niqqud class so the pattern matches pointed
text.  Non-Hebrew characters (including regex metacharacters) are left
untouched, so this is safe to apply to an already-compiled regexp."
  (let (out)
    (dolist (ch (string-to-list regexp))
      (cond
       ((seforim--niqqud-char-p ch) nil)          ; strip niqqud from query
       ((seforim--letter-p ch)
        (push (char-to-string ch) out)
        (push seforim--niqqud-opt out))
       (t (push (char-to-string ch) out))))
    (apply #'concat (nreverse out))))

(defun seforim--build-niqqud-regex (query)
  "Build a regex from QUERY matching QUERY regardless of niqqud."
  (seforim--inject-niqqud query))

(defun seforim--niqqud-regexp-compiler (input type ignore-case)
  "Consult regexp compiler that ignores Hebrew niqqud.
Wraps the default `consult--regexp-compiler': compiles INPUT normally,
then rewrites each produced regexp so Hebrew letters tolerate niqqud.
INPUT, TYPE and IGNORE-CASE are passed through unchanged."
  (pcase-let ((`(,regexps . ,hl)
               (funcall (default-value 'consult--regexp-compiler)
                        input type ignore-case)))
    (cons (mapcar #'seforim--inject-niqqud regexps) hl)))

;; ---------------------------------------------------------------------------
;; Path Helpers
;; ---------------------------------------------------------------------------

(defun seforim--base-dir ()
  "Return the normalized, existing base directory.
Falls back to auto-detection if `seforim-directory' does not exist,
so a wrong casing in the custom value never breaks the commands."
  (let ((d (expand-file-name seforim-directory)))
    (file-name-as-directory
     (if (file-directory-p d) d (or (seforim--default-directory) d)))))

(defun seforim--format-path (path)
  "Shorten PATH for display, showing last 3 components."
  (let* ((base (seforim--base-dir))
         (rel  (file-relative-name (expand-file-name path) base))
         (parts (split-string rel "/")))
    (if (>= (length parts) 3)
        (mapconcat #'identity (last parts 3) "/")
      rel)))

(defun seforim--list-subdirs ()
  "List immediate subdirectories of the seforim base directory."
  (let ((base (seforim--base-dir)))
    (when (file-directory-p base)
      (seq-filter (lambda (f)
                    (and (file-directory-p f)
                         (not (member (file-name-nondirectory f) '("." "..")))))
                  (directory-files base t)))))
